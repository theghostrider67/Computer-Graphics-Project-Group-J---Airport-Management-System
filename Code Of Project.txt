#include <windows.h>
#include <GL/glut.h>
#include <cmath>
#include <cstdlib> // Included for generating random numbers (weather/stars)
#include <ctime>   // Included to seed the random number generator with current time

// GLOBAL STATE VARIABLES

// Tracks which airport is currently being displayed (1 = Rafi, 2 = Tasin, 3 = Fahim)
int currentAirport = 1;

// Tracks the current phase of the flight:
// 0 = Takeoff from Rafi, 1 = Landing at Tasin, 2 = Takeoff from Tasin, 3 = Landing at Fahim
int flightStage = 0;

// Airplane position and rotation variables
float planeX = -1.2f;    // Horizontal position of the plane (starts off-screen left)
float planeY = -0.7f;    // Vertical position of the plane (starts on the runway)
float planeAngle = 0.0f; // Pitch angle of the plane (tilt up/down)
float cloudX = -1.5f;    // Horizontal position for the background clouds

//Speed & Control Flags
float baseSpeed = 0.012f;  // The default reference speed used to calculate takeoff/landing curves
float planeSpeed = 0.012f; // The current speed of the plane, adjustable by the user
bool isStopped = false;    // Boolean flag to track if the plane is halted on the runway

// Environmental Toggles(Day/Night & Weather)
bool isNight = false;      // Toggles night mode (dark sky, stars)
bool isRaining = false;    // Toggles rain particle effect
bool isSnowing = false;    // Toggles snow particle effect

//  Sliding Text & Window State 
float signScroll = 0.45f;  // Controls the horizontal scrolling position of the text on the tower
int winW = 1200;           // Tracks current window width (needed for glScissor text clipping)
int winH = 650;            // Tracks current window height (needed for glScissor text clipping)


// WEATHER & ENVIRONMENT PARTICLE SYSTE

#define MAX_PARTICLES 300 // Maximum number of rain/snow particles on screen

// Structure to define a single weather particle (raindrop or snowflake)
struct Particle {
    float x, y;    // Current coordinates
    float speed;   // How fast it falls
    float drift;   // Horizontal drift (used for snow swaying)
};
Particle weather[MAX_PARTICLES]; // Array holding all weather particles

// Structure to define a single background star
struct Star {
    float x, y; // Coordinates of the star
};
Star stars[100]; // Array holding 100 stars

// Function to initialize the random positions and speeds of stars and weather particles
void initEnvironment() {
    srand((unsigned int)time(0)); // Seed random generator so it's different every time the app runs

    // Initialize Weather Particles with random starting positions and speeds
    for (int i = 0; i < MAX_PARTICLES; i++) {
        weather[i].x = ((rand() % 500) / 100.0f) - 2.5f; // Random X between -2.5 and 2.5
        weather[i].y = ((rand() % 200) / 100.0f) - 1.0f; // Random Y between -1.0 and 1.0
        weather[i].speed = ((rand() % 100) / 2000.0f) + 0.02f; // Random falling speed
        weather[i].drift = ((rand() % 100) / 10000.0f);        // Random side-to-side drift
    }

    // Initialize Stars with random positions in the upper half of the screen
    for (int i = 0; i < 100; i++) {
        stars[i].x = ((rand() % 500) / 100.0f) - 2.5f; // Random X spread across the screen
        stars[i].y = ((rand() % 100) / 100.0f);        // Random Y from 0.0 to 1.0 (sky area)
    }
}

// MATH & DRAWING UTILITIES

// Helper function to draw a filled circle using trigonometry
void drawCircle(float x, float y, float radius) {
    glBegin(GL_TRIANGLE_FAN);
    for (int i = 0; i <= 100; i++) {
        float angle = 2.0f * 3.1415926f * i / 100; // Calculate angle in radians
        glVertex2f(x + radius * cosf(angle), y + radius * sinf(angle)); // Calculate vertex
    }
    glEnd();
}

// Helper function to draw a fluffy cloud by grouping three circles together
void drawCloud(float x, float y, float scale, float r = 1.0f, float g = 1.0f, float b = 1.0f) {
    glColor3f(r, g, b); // Set cloud color (allows for tinted clouds at sunset/night)
    drawCircle(x, y, 0.1f * scale);                            // Left puff
    drawCircle(x + 0.08f * scale, y + 0.03f * scale, 0.12f * scale); // Top/Center puff
    drawCircle(x + 0.15f * scale, y, 0.1f * scale);            // Right puff
}

// Helper function to render a text string at a specific (x, y) coordinate
void drawText(float x, float y, const char* text) {
    glRasterPos2f(x, y); // Set the starting position for the text
    while (*text) {
        glutBitmapCharacter(GLUT_BITMAP_HELVETICA_12, *text++); // Draw each character one by one
    }
}

// VISUAL AIRPORT ELEMENTS

// Draws the electronic signboard on the control tower with scrolling text
void drawSignboard(float centerX, float centerY, const char* text) {
    float boxWidth = 0.9f;
    float height = 0.12f;

    // 1. Draw the dark background frame of the signboard
    glColor3f(0.1f, 0.1f, 0.1f);
    glRectf(centerX - boxWidth/2.0f, centerY - height/2.0f, centerX + boxWidth/2.0f, centerY + height/2.0f);

    // 2. Draw the silver border/trim around the signboard
    glColor3f(0.7f, 0.7f, 0.75f);
    glLineWidth(2);
    glBegin(GL_LINE_LOOP);
    glVertex2f(centerX - boxWidth/2.0f, centerY - height/2.0f);
    glVertex2f(centerX + boxWidth/2.0f, centerY - height/2.0f);
    glVertex2f(centerX + boxWidth/2.0f, centerY + height/2.0f);
    glVertex2f(centerX - boxWidth/2.0f, centerY + height/2.0f);
    glEnd();

    // 3. Draw the sliding text, restricting it to stay inside the signboard using glScissor
    glColor3f(1.0f, 1.0f, 1.0f);
    glEnable(GL_SCISSOR_TEST); // Enable clipping

    // Convert standard OpenGL coordinates (-1 to 1) to screen pixel coordinates for glScissor
    int pixX = (int)(((centerX - boxWidth/2.0f) + 1.5f) / 3.0f * winW);
    int pixY = (int)(((centerY - height/2.0f) + 1.0f) / 2.0f * winH);
    int pixW = (int)(boxWidth / 3.0f * winW);
    int pixH = (int)(height / 2.0f * winH);

    glScissor(pixX, pixY, pixW, pixH); // Define the clipping rectangle

    // Draw the text using the scrolling variable
    float startX = centerX + signScroll;
    drawText(startX, centerY - 0.02f, text);

    glDisable(GL_SCISSOR_TEST); // Disable clipping so the rest of the scene draws normally
}

// Draws the flag of Bangladesh on a flagpole
void drawBDFlag(float poleX, float poleY, float scale) {
    // Flag pole
    glColor3f(0.8f, 0.8f, 0.8f);
    glLineWidth(3);
    glBegin(GL_LINES);
    glVertex2f(poleX, poleY);
    glVertex2f(poleX, poleY + 0.2f * scale);
    glEnd();
    glLineWidth(1);

    // Green rectangular background
    glColor3f(0.0f, 0.42f, 0.24f);
    glRectf(poleX, poleY + 0.08f * scale, poleX + 0.2f * scale, poleY + 0.2f * scale);

    // Red circle in the middle
    glColor3f(0.86f, 0.16f, 0.24f);
    drawCircle(poleX + 0.09f * scale, poleY + 0.14f * scale, 0.04f * scale);
}

// Reusable function to draw a modern, multi-level airport terminal building
// Allows passing custom RGB colors for the building walls and the glass windows
void drawModernTerminal(float r, float g, float b, float glassR, float glassG, float glassB) {
    // Main ground floor block
    glColor3f(r, g, b);
    glRectf(-1.4f, -0.7f, 0.1f, 0.0f);

    // Upper level block (calculated to be slightly lighter in color for depth)
    float r2 = r + 0.1f; if(r2 > 1.0f) r2 = 1.0f;
    float g2 = g + 0.1f; if(g2 > 1.0f) g2 = 1.0f;
    float b2 = b + 0.1f; if(b2 > 1.0f) b2 = 1.0f;
    glColor3f(r2, g2, b2);
    glRectf(-1.2f, 0.0f, -0.2f, 0.2f);

    // Draw rows of glass windows
    glColor3f(glassR, glassG, glassB);
    for(float wx = -1.35f; wx < 0.0f; wx += 0.2f) glRectf(wx, -0.5f, wx + 0.15f, -0.2f);
    for(float wx = -1.15f; wx < -0.3f; wx += 0.2f) glRectf(wx, 0.05f, wx + 0.15f, 0.15f);

    // Draw entrance doors at the bottom
    glColor3f(0.1f, 0.1f, 0.1f);
    glRectf(-1.1f, -0.7f, -1.0f, -0.55f);
    glRectf(-0.3f, -0.7f, -0.2f, -0.55f);

    // Draw Jet bridges (tunnels connecting terminal to planes)
    glColor3f(0.5f, 0.5f, 0.5f);
    glRectf(0.1f, -0.6f, 0.3f, -0.5f);
    glRectf(0.3f, -0.65f, 0.4f, -0.45f);
}

// Draws the static ground (grass) and the asphalt runway at the bottom of the screen
void drawGroundAndRunway() {
    // Set different grass colors depending on which airport is currently visible
    if (currentAirport == 1) glColor3f(0.3f, 0.65f, 0.3f);      // Bright grass
    else if (currentAirport == 2) glColor3f(0.2f, 0.45f, 0.2f); // Darker, industrial grass
    else glColor3f(0.25f, 0.5f, 0.25f);                         // Standard green grass

    // Draw the grass block
    glRectf(-2.5f, -1.0f, 2.5f, -0.3f);

    // Draw the dark grey asphalt runway on top of the grass
    glColor3f(0.18f, 0.18f, 0.18f);
    glRectf(-2.5f, -0.92f, 2.5f, -0.65f);

    // Draw the white dashed center lines on the runway
    glColor3f(1.0f, 1.0f, 1.0f);
    for (float i = -2.4f; i < 2.5f; i += 0.6f) glRectf(i, -0.8f, i + 0.3f, -0.77f);
}

// Function to draw Airport #1 (Rafi's International Airport)
void drawRafiAirport() {
    // Set sky color based on Day/Night toggle
    if (isNight) glClearColor(0.05f, 0.1f, 0.15f, 1.0f);
    else glClearColor(0.35f, 0.65f, 0.95f, 1.0f);

    // Draw distant mountain range
    glColor3f(0.4f, 0.5f, 0.6f);
    glBegin(GL_TRIANGLES);
    glVertex2f(-1.5f, -0.3f); glVertex2f(-0.8f, 0.2f); glVertex2f(-0.1f, -0.3f);
    glVertex2f(0.5f, -0.3f); glVertex2f(1.2f, 0.3f); glVertex2f(1.9f, -0.3f);
    glEnd();

    drawGroundAndRunway();
    drawModernTerminal(0.8f, 0.82f, 0.85f, 0.3f, 0.6f, 0.8f); // Terminal with light grey walls, blue glass

    // Draw the unique round-shaped Control Tower
    float tx = 0.6f, ty = 0.25f;
    glColor3f(0.7f, 0.72f, 0.75f); glRectf(tx - 0.1f, -0.7f, tx + 0.1f, 0.1f); // Base shaft
    glColor3f(0.6f, 0.62f, 0.65f); glRectf(tx - 0.02f, -0.7f, tx + 0.02f, 0.1f); // Center support line
    glColor3f(0.85f, 0.88f, 0.9f); drawCircle(tx, ty, 0.35f); // Outer globe
    glColor3f(0.2f, 0.4f, 0.65f); drawCircle(tx, ty, 0.28f);  // Inner glass globe

    // Add a glowing radar screen effect inside the glass globe
    glColor3f(0.0f, 0.8f, 0.5f); glPointSize(3.0f); glBegin(GL_POINTS);
    glVertex2f(tx - 0.05f, ty - 0.05f); glVertex2f(tx + 0.05f, ty - 0.15f); glVertex2f(tx, ty + 0.05f); glEnd();

    // Draw geodesic structural lines on the globe
    glColor3f(0.4f, 0.4f, 0.45f); glLineWidth(2); glBegin(GL_LINES);
    glVertex2f(tx, ty - 0.35f); glVertex2f(tx, ty + 0.35f);
    glVertex2f(tx - 0.25f, ty - 0.2f); glVertex2f(tx + 0.25f, ty + 0.2f);
    glVertex2f(tx + 0.25f, ty - 0.2f); glVertex2f(tx - 0.25f, ty + 0.2f); glEnd();

    // Tower Antenna and Flag
    glColor3f(0.2f, 0.2f, 0.2f); glBegin(GL_LINES); glVertex2f(tx, ty + 0.35f); glVertex2f(tx, ty + 0.55f); glEnd();
    drawBDFlag(tx, ty + 0.45f, 1.2f);

    // Draw the named signboard attached to the tower
    drawSignboard(tx, -0.1f, "Rafi's International Airport");

    // Only draw fluffy clouds if it is daytime
    if (!isNight) drawCloud(cloudX, 0.7f, 1.0f, 1.0f, 1.0f, 1.0f);
}

// ENVIRONMENT OVERLAYS (NIGHT & WEATHER)

// Renders the stars array if it is night time
void drawStars() {
    if (!isNight) return; // Exit early if it's day
    glColor3f(1.0f, 1.0f, 1.0f);
    glPointSize(2.0f);
    glBegin(GL_POINTS);
    for (int i = 0; i < 100; i++) glVertex2f(stars[i].x, stars[i].y);
    glEnd();
}

// Draws a translucent dark quad over the whole screen to dim the colors during night mode
void drawNightOverlay() {
    if (!isNight) return;
    glEnable(GL_BLEND); // Enable opacity blending
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    glColor4f(0.0f, 0.05f, 0.15f, 0.4f); // Transparent dark blue tint
    glRectf(-3.0f, -3.0f, 3.0f, 3.0f);   // Cover entire screen
    glDisable(GL_BLEND); // Turn off blending
}

// Renders rain lines or snow points based on active toggles
void drawWeather() {
    if (!isRaining && !isSnowing) return; // Exit early if weather is clear

    if (isRaining) {
        glColor3f(0.6f, 0.7f, 0.8f); // Set to transparent greyish-blue
        glLineWidth(1.5f);
        glBegin(GL_LINES);
        for (int i = 0; i < MAX_PARTICLES; i++) {
            glVertex2f(weather[i].x, weather[i].y); // Rain drop start
            glVertex2f(weather[i].x - 0.02f, weather[i].y - 0.06f); // Rain drop end (slanted to simulate wind)
        }
        glEnd();
        glLineWidth(1.0f);
    } else if (isSnowing) {
        glColor3f(1.0f, 1.0f, 1.0f); // Set to pure white
        glPointSize(3.0f);
        glBegin(GL_POINTS);
        for (int i = 0; i < MAX_PARTICLES; i++) {
            glVertex2f(weather[i].x, weather[i].y); // Draw snowflake as a point
        }
        glEnd();
    }
}

// THE AIRPLANE

// Draws the main aircraft, taking into account its current X, Y, and rotation angle
void drawPlane() {
    glPushMatrix(); // Save current matrix state

    // Apply dynamic transformations
    glTranslatef(planeX, planeY, 0.0f);       // Move plane to current position
    glRotatef(planeAngle, 0.0f, 0.0f, 1.0f);  // Tilt plane based on takeoff/landing
    glScalef(1.4f, 1.4f, 1.0f);               // Scale plane size up slightly

    // Draw Main Plane Body (Fuselage)
    glColor3f(0.95f, 0.95f, 0.95f);
    glBegin(GL_POLYGON);
    glVertex2f(-0.1f, -0.02f); glVertex2f(0.08f, -0.02f); glVertex2f(0.12f, 0.0f);
    glVertex2f(0.08f, 0.02f); glVertex2f(-0.1f, 0.02f);
    glEnd();

    // Draw decorative yellow stripe
    glColor3f(1.0f, 0.8f, 0.0f);
    glRectf(-0.1f, -0.005f, 0.09f, 0.005f);

    // Draw Wings and Tail Fin
    glColor3f(0.1f, 0.3f, 0.8f);
    glBegin(GL_TRIANGLES);
    glVertex2f(0.02f, 0.02f); glVertex2f(-0.04f, 0.1f); glVertex2f(-0.04f, 0.02f);    // Top Wing
    glVertex2f(0.02f, -0.02f); glVertex2f(-0.04f, -0.1f); glVertex2f(-0.04f, -0.02f); // Bottom Wing
    glVertex2f(-0.1f, 0.02f); glVertex2f(-0.12f, 0.06f); glVertex2f(-0.06f, 0.02f);   // Tail
    glEnd();

    // Draw Landing Gear (Wheels)
    glColor3f(0.1f, 0.1f, 0.1f);
    drawCircle(-0.06f, -0.025f, 0.015f); // Back wheel
    drawCircle(0.06f, -0.025f, 0.01f);   // Front wheel

    glPopMatrix(); // Restore matrix state (so transformations don't affect other objects)
}

// USER INPUT HANDLERS

// Handles standard ASCII keyboard inputs (letters, numbers, etc.)
void handleKeypress(unsigned char key, int x, int y) {
    if (key == 's' || key == 'S') {
        // Halt plane ONLY if it is currently grounded on the runway
        if (planeY <= -0.69f) isStopped = true;
    } else if (key == 'w' || key == 'W') {
        // Resume flight movement
        isStopped = false;
    } else if (key == 'n' || key == 'N') {
        // Toggle Night Mode ON
        isNight = true;
    } else if (key == 'd' || key == 'D') {
        // Toggle Day Mode ON (Night Mode OFF)
        isNight = false;
    } else if (key == 'r' || key == 'R') {
        // Toggle Rain. If turning rain ON, ensure snow is turned OFF.
        isRaining = !isRaining;
        isSnowing = false;
    } else if (key == 't' || key == 'T') {
        // Toggle Snow. If turning snow ON, ensure rain is turned OFF.
        isSnowing = !isSnowing;
        isRaining = false;
    }
    glutPostRedisplay(); // Request a screen redraw to reflect changes immediately
}

// Handles special non-ASCII keyboard inputs (Arrow keys, F-keys, etc.)
void handleSpecialKeypress(int key, int x, int y) {
    switch (key) {
        case GLUT_KEY_UP:
            // Increase plane speed
            planeSpeed += 0.002f;
            break;
        case GLUT_KEY_DOWN:
            // Decrease plane speed
            planeSpeed -= 0.002f;
            // Cap minimum speed so the plane doesn't stop or fly backward via arrows
            if (planeSpeed < 0.002f) planeSpeed = 0.002f;
            break;
    }
    glutPostRedisplay();
}

// MAIN UPDATE / ANIMATION LOOP

// Called periodically by glutTimerFunc to update positions of moving objects
void update(int value) {
    // 1. Update Clouds (Constant horizontal scroll)
    cloudX += 0.003f;
    if (cloudX > 2.2f) cloudX = -2.2f; // Wrap around screen

    // 2. Update Signboard Text (Constant horizontal scroll)
    signScroll -= 0.006f;
    if (signScroll < -1.2f) signScroll = 0.45f; // Wrap back to right edge

    // 3. Update Weather Particles (Falling logic)
    if (isRaining || isSnowing) {
        for (int i = 0; i < MAX_PARTICLES; i++) {
            // Rain falls faster than snow
            weather[i].y -= (isRaining ? weather[i].speed * 1.0f : weather[i].speed * 0.2f);

            // Apply horizontal swaying motion if it is snowing
            if (isSnowing) weather[i].x += (sin(weather[i].y * 10.0f) * weather[i].drift);

            // If particle hits the bottom, reset it to the top with a new random X
            if (weather[i].y < -1.0f) {
                weather[i].y = 1.0f;
                weather[i].x = ((rand() % 500) / 100.0f) - 2.5f;
            }
        }
    }

    // 4. Update Automatic Flight Path (If not manually stopped)
    if (!isStopped) {
        planeX += planeSpeed; // Move forward

        // Ratio ensures the takeoff/landing curve remains identical even if speed changes
        float ratio = planeSpeed / baseSpeed;

        // State Machine handling the sequence of the 3-airport journey
        switch (flightStage) {
            case 0: // DEPARTING Rafi's Airport
                currentAirport = 1;
                if (planeX > 0.1f) {
                    planeY += 0.008f * ratio; // Lift off
                    if (planeAngle < 20.0f) planeAngle += 0.5f * ratio; // Nose up
                }
                // When off screen right, transition to Tasin arrival stage
                if (planeX > 2.0f) { flightStage = 1; planeX = -2.0f; planeY = 0.8f; planeAngle = -15.0f; }
                break;

            case 1: // ARRIVING Tasin's Airport
                currentAirport = 2;
                if (planeY > -0.7f) {
                    planeY -= 0.008f * ratio; // Descend
                    if (planeAngle < 0.0f) planeAngle += 0.2f * ratio; // Level nose
                } else {
                    planeY = -0.7f; planeAngle = 0.0f; // Grounded
                    if (planeX > 0.5f) flightStage = 2; // Begin takeoff phase after taxiing
                }
                break;

            case 2: // DEPARTING Tasin's Airport
                if (planeX > 0.8f) {
                    planeY += 0.01f * ratio; // Lift off steeper
                    if (planeAngle < 20.0f) planeAngle += 0.8f * ratio; // Nose up faster
                }
                // When off screen right, transition to Fahim arrival stage
                if (planeX > 2.0f) { flightStage = 3; planeX = -2.0f; planeY = 0.8f; planeAngle = -15.0f; }
                break;

            case 3: // ARRIVING Fahim's Airport
                currentAirport = 3;
                if (planeY > -0.7f) {
                    planeY -= 0.008f * ratio; // Descend
                    if (planeAngle < 0.0f) planeAngle += 0.2f * ratio; // Level nose
                } else {
                    planeY = -0.7f; planeAngle = 0.0f; // Grounded
                }
                // When off screen right, loop back to the beginning at Rafi's
                if (planeX > 2.0f) { flightStage = 0; planeX = -1.2f; planeY = -0.7f; planeAngle = 0.0f; }
                break;
        }
    }

    glutPostRedisplay(); // Request a redraw to show updated frame
    glutTimerFunc(16, update, 0); // ~60 FPS update loop
}

// CORE GLUT FUNCTIONS (DISPLAY, RESHAPE, MAIN)

// Master rendering function called every frame
void display() {
    glClear(GL_COLOR_BUFFER_BIT); // Clear previous frame
    glLoadIdentity();             // Reset coordinate system

    // 1. Draw the Background / Environment / Buildings based on current stage
    if (currentAirport == 1) drawRafiAirport();
    else if (currentAirport == 2) drawTasinAirport();
    else drawFahimAirport();

    // 2. Draw active Environmental Overlays
    drawStars();
    drawNightOverlay();
    drawWeather();

    // 3. Draw the Plane on top of everything
    drawPlane();

    glutSwapBuffers(); // Push drawn frame to the screen (Double Buffering)
}

// Handles window resizing to prevent aspect ratio distortion
void reshape(int w, int h) {
    winW = w; // Update global tracking for glScissor
    winH = h;
    glViewport(0, 0, w, h);       // Reset the viewport to cover the new window size
    glMatrixMode(GL_PROJECTION);  // Switch to projection matrix
    glLoadIdentity();
    gluOrtho2D(-1.5, 1.5, -1.0, 1.0); // Set 2D orthographic viewing box boundaries
    glMatrixMode(GL_MODELVIEW);   // Return to modelview for drawing
}

int main(int argc, char** argv) {
    // 1. Setup GLUT
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB);
    glutInitWindowSize(1200, 650);
    glutCreateWindow("Triple Airport System - Weather & Night Mode");

    // 2. Initialize application specific data
    initEnvironment();

    // 3. Register callback functions
    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutTimerFunc(0, update, 0);
    glutKeyboardFunc(handleKeypress);        // Register standard keys
    glutSpecialFunc(handleSpecialKeypress);  // Register arrow keys

    // 4. Enter main rendering loop
    glutMainLoop();
    return 0;
}
